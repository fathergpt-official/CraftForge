    <script>
        function switchTab(evt, tabId) {
            let contents = document.querySelectorAll('.panel-tab-content');
            contents.forEach(c => c.style.display = 'none');

            let items = document.querySelectorAll('.panel-menu li');
            items.forEach(i => i.classList.remove('active'));

            document.getElementById(tabId).style.display = 'block';
            evt.currentTarget.classList.add('active');
        }

        function copyAddress() {
            let copyText = document.getElementById("serverAddressInput");
            copyText.select();
            navigator.clipboard.writeText(copyText.value);
            alert("Server address copied to clipboard!");
        }

        function toggleServerState(state) {
            let badge = document.getElementById("serverStatusBadge");
            if(state === 'start') {
                badge.className = "badge-online";
                badge.innerHTML = '<i class="fa-solid fa-circle"></i> Online (24/7 Uptime)';
                appendConsoleLog("[Server thread/INFO]: Server started successfully!", "success");
            } else if(state === 'stop') {
                badge.className = "badge-offline";
                badge.innerHTML = '<i class="fa-solid fa-circle"></i> Offline';
                appendConsoleLog("[Server thread/INFO]: Stopping server...", "error");
            } else if(state === 'restart') {
                badge.className = "badge-online";
                badge.innerHTML = '<i class="fa-solid fa-circle"></i> Restarting...';
                appendConsoleLog("[Server thread/INFO]: Restarting server instances...", "info");
            }
        }

        function appendConsoleLog(text, type) {
            let term = document.getElementById("consoleTerminal");
            let p = document.createElement("p");
            p.className = type === 'success' ? 'log-success' : type === 'error' ? 'log-error' : 'log-info';
            p.innerText = text;
            term.appendChild(p);
            term.scrollTop = term.scrollHeight;
        }

        function handleConsoleCommand(e) {
            if(e.key === 'Enter') {
                sendConsoleCommand();
            }
        }

        function sendConsoleCommand() {
            let input = document.getElementById("consoleCommandInput");
            if(input.value.trim() !== "") {
                appendConsoleLog("> " + input.value, "chat");
                if(input.value.startsWith("say ")) {
                    appendConsoleLog("[Server] " + input.value.substring(4), "chat");
                } else {
                    appendConsoleLog("[Server thread/INFO]: Executed command: " + input.value, "info");
                }
                input.value = "";
            }
        }

        function toggleOnlineMode() {
            let toggle = document.getElementById("onlineModeToggle");
            if(toggle.checked) {
                alert("Cracked (Offline Mode) Enabled! TLauncher & non-premium players can now join.");
            } else {
                alert("Cracked Mode Disabled! Only official premium Mojang/Microsoft accounts can join.");
            }
        }
    </script>
